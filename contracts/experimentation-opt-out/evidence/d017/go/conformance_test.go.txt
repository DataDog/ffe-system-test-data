package openfeature

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"testing"
	"time"

	of "github.com/open-feature/go-sdk/openfeature"
)

// The source overlay records production entry into excludes and
// evaluateAllocation; no observation is derived from the requested mode.
var d017Trace struct {
	checks   int
	excluded []bool
}

func TestLeanD017OperationalConformance(t *testing.T) {
	f, err := os.Open(os.Getenv("LEAN_D017_CASES"))
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()
	scanner := bufio.NewScanner(f)
	rows := 0
	for scanner.Scan() {
		var row struct {
			ID   string `json:"id"`
			Args struct {
				Mode         string `json:"mode"`
				Attribute    string `json:"attribute"`
				CheckResults []bool `json:"checkResults"`
			} `json:"args"`
			Expected struct {
				Error           string `json:"error"`
				Excluded        []bool `json:"excluded"`
				ExclusionChecks int    `json:"exclusionChecks"`
			} `json:"expected"`
		}
		if err := json.Unmarshal(scanner.Bytes(), &row); err != nil {
			t.Fatal(err)
		}
		rows++
		t.Run(row.ID, func(t *testing.T) {
			modes := map[string]ExperimentationOptOutMode{
				"disabled":                ExperimentationOptOutDisabled,
				"immediate":               ExperimentationOptOutImmediate,
				"future_experiments_only": ExperimentationOptOutFutureExperimentsOnly,
			}
			mode, ok := modes[row.Args.Mode]
			if !ok {
				t.Fatal("unknown mode")
			}
			activation := time.Date(2026, 10, 19, 12, 0, 0, 0, time.UTC)
			allocations := make([]*allocation, 0, len(row.Args.CheckResults))
			for i, check := range row.Args.CheckResults {
				// doLog=false retains; doLog=true with equal activation excludes
				// in either enabled mode. Empty splits ensure every allocation is
				// visited, independently of the exclusion outcome.
				allocations = append(allocations, &allocation{
					Key: fmt.Sprint(i), DoLog: new(check), RandomizationActivatedAt: &activation,
				})
			}
			p := newDatadogProvider(ProviderConfig{ExperimentationOptOutMode: mode})
			t.Cleanup(p.Shutdown)
			p.configuration = &universalFlagsConfiguration{Flags: map[string]*flag{
				"trace": {Key: "trace", Enabled: true, VariationType: valueTypeBoolean, Allocations: allocations},
			}}
			attrs := of.FlattenedContext{of.TargetingKey: "lean-subject"}
			switch row.Args.Attribute {
			case "valid":
				attrs[experimentationOptOutAtAttribute] = activation.Format(time.RFC3339Nano)
			case "invalid":
				attrs[experimentationOptOutAtAttribute] = false
			case "absent":
			default:
				t.Fatal("unknown attribute")
			}
			d017Trace.checks = 0
			d017Trace.excluded = []bool{}
			result := p.BooleanEvaluation(context.Background(), "trace", false, attrs)
			gotError := string(result.ResolutionDetail().ErrorCode)
			if gotError == "" {
				gotError = "NONE"
			}
			if gotError != row.Expected.Error || !reflect.DeepEqual(d017Trace.excluded, row.Expected.Excluded) || d017Trace.checks != row.Expected.ExclusionChecks {
				t.Fatalf("error=%s excluded=%v exclusionChecks=%d; want error=%s excluded=%v exclusionChecks=%d", gotError, d017Trace.excluded, d017Trace.checks, row.Expected.Error, row.Expected.Excluded, row.Expected.ExclusionChecks)
			}
			if result.Value || result.Variant != "" {
				t.Fatalf("empty splits unexpectedly selected a variant: %+v", result)
			}
			t.Logf("observed exclusionChecks=%d excluded=%v error=%s", d017Trace.checks, d017Trace.excluded, gotError)
		})
	}
	if err := scanner.Err(); err != nil {
		t.Fatal(err)
	}
	if rows != 36 {
		t.Fatalf("expected exactly 36 pinned rows, got %d", rows)
	}
}
