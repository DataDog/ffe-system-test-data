def output_instant(normalized):
 """Representation-only conversion after production has accepted and normalized input.
 Python datetime is proleptic Gregorian; shift by one 400-year cycle for year zero.
 Invalid production normalization remains an accepted observation and fails comparison.
 """
 m=re.fullmatch(r'(\d+)-(\d\d)-(\d\d)T(\d\d):(\d\d):(\d\d)(?:\.(\d+))?Z',normalized)
 if not m: raise ValueError('production output is not canonical UTC')
 year,month,day,hour,minute,second=map(int,m.groups()[:6])
 if year>9999: raise ValueError('production output exceeds four-digit UTC year')
 represented=datetime.datetime(400 if year==0 else year,month,day,hour,minute,second)
 delta=represented-datetime.datetime(1970,1,1)
 seconds=delta.days*86400+delta.seconds-(146097*86400 if year==0 else 0)
 return {'accepted':True,'seconds':seconds,'nanos':int((m.group(7) or '').ljust(9,'0'))}
